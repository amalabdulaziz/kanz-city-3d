<input id="cmd" placeholder="اكتبي أمرك هنا..." />

<button onclick="run()">تنفيذ</button>


<script>

let state = "door";


function run(){

  let c = document.getElementById("cmd").value.trim();

  let s = document.getElementById("story");


  if(state==="door"){

    if(c==="ادخل" || c==="ادخلي"){

      s.innerText="الممر مظلم… تسمعين صريرًا بعيدًا. ماذا تفعلين؟ (تفقد / اصعد / خروج)";

      state="hall";

    } else {

      s.innerText="حاولي الأمر: (ادخل)";

    }

  }


  else if(state==="hall"){

    if(c==="تفقد"){

      s.innerText="تجدين ملفًا قديمًا باسمك… لماذا هو هنا؟ (اصعد / خروج)";

    }

    else if(c==="اصعد"){

      s.innerText="الطابق العلوي أبرد… باب غرفة العمليات مفتوح. (تفقد / انزل / خروج)";

      state="upper";

    }

    else if(c==="خروج"){

      s.innerText="تهربين خارج المستشفى… لكن الحلم لم ينتهِ!";

      state="end";

    } else {

      s.innerText="الأوامر المتاحة: تفقد / اصعد / خروج";

    }

  }


  else if(state==="upper"){

    if(c==="تفقد"){

      s.innerText="تجدين رسالة: (لا تعودي). فجأة تسمعين خطوات. (انزل / خروج)";

    }

    else if(c==="انزل"){

      s.innerText="ترجعين للممر… القلب يخفق بقوة. (تفقد / خروج)";

      state="hall";

    }

    else if(c==="خروج"){

      s.innerText="تستيقظين من الكابوس… أو هكذا تظنين.";

      state="end";

    } else {

      s.innerText="الأوامر المتاحة: تفقد / انزل / خروج";

    }

  }


  document.getElementById("cmd").value="";

}

</script>
