let audioCtx = null;
function stepSound(now, fast) {
  if (!soundEnabled) return;
  try {
    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    if (audioCtx.state === 'suspended') return;
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'triangle';
    osc.frequency.value = fast ? 175 : 135;
    gain.gain.setValueAtTime(0.025, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 0.06);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime + 0.065);
  } catch {}
}

function updateTraffic(dt) {
  for (const c of movingTraffic) {
    const here = c.from + (c.to - c.from) * c.offset;
    const nearPlayer = c.axis === 'z' ? Math.hypot(c.lane - pos.x, here - pos.z) : Math.hypot(here - pos.x, c.lane - pos.z);
    if (nearPlayer >= 4.3) {
      let dist = c.to - c.from;
      c.offset = (c.offset + c.speed * dt / dist + 2) % 1;
    }
    const k = c.from + (c.to - c.from) * c.offset;
    c.group.position.set(c.axis === 'z' ? c.lane : k, .06, c.axis === 'z' ? k : c.lane);
  }
}

function walkable(x, z) {
  if (Math.abs(x) > 68 || Math.abs(z) > 68) return false;
  for (const c of collision) {
    if (x > c.minX - .45 && x < c.maxX + .45 && z > c.minZ - .45 && z < c.maxZ + .45) return false;
  }
  for (const rc of roundCollision) {
    if (Math.hypot(x - rc.x, z - rc.z) < rc.r + .42) return false;
  }
  return true;
}

function near(a, b, dist) {
  return Math.hypot(a.x - b.x, a.z - b.z) <= dist;
}

function nearestInteraction() {
  if (keyGroup.visible && near(pos, keyGroup.position, 2.2)) {
    return { kind: 'key', label: 'التقطي المفتاح الذهبي' };
  }
  const ch = chests[0];
  if (ch && !ch.opened && near(pos, ch, 2.6)) {
    return { kind: 'chest', label: hasKey ? 'افتحي الصندوق الأسطوري' : 'الصندوق مغلق (يحتاج المفتاح)' };
  }
  for (const it of interactables) {
    if (it.kind === 'flower' && it.watered) continue;
    if (near(pos, it, 2.4)) return it;
  }
  return null;
}

function triggerInteraction() {
  if (activity) {
    activity = null;
    toast('نهضتِ بنجاح');
    return;
  }
  const item = nearestInteraction();
  if (!item) return;

  if (item.kind === 'key') {
    keyGroup.visible = false;
    hasKey = true;
    data.gate = true;
    toast('🗝️ حصلتِ على المفتاح الذهبي!');
    syncState();
  } else if (item.kind === 'chest') {
    if (!hasKey) {
      toast('🔒 تحتاجين إلى المفتاح الذهبي أولاً!');
      return;
    }
    const ch = chests[0];
    ch.opened = true;
    chestOpened = true;
    coins += 100;
    syncState();
    setTimeout(() => {
      if ($('finalScore')) $('finalScore').textContent = `💎 الجواهر: ${gems}/5 · 🪙 العملات: ${coins}`;
      if ($('winOverlay')) $('winOverlay').classList.add('open');
    }, 650);
  } else if (item.kind === 'flower') {
    if (waterCharges <= 0 && ammo <= 0) {
      toast('💧 لا يوجد ماء كافٍ الآن، جرّبي زر الرش أو السوبر ماركت');
      return;
    }
    if (waterCharges > 0) waterCharges--;
    else ammo = Math.max(0, ammo - 2);
    item.watered = true;
    item.blossom.visible = true;
    item.blossom.scale.setScalar(2.4);
    flowersGrown++;
    coins += 20;
    toast(`🌸 أزهرت الحديقة! (${flowersGrown}/3) +20 عملة`);
    syncState();
  } else if (item.kind === 'seat' || item.kind === 'swing') {
    activity = { kind: item.kind, x: item.x, z: item.z };
    toast(item.kind === 'swing' ? '🛝 استمتعي بالمرجوحة!' : '🪑 استراحة لطيفة على المقعد');
  } else if (item.kind === 'snack') {
    if (coins >= 20) {
      coins -= 20;
      snackBoostUntil = performance.now() / 1000 + 18;
      ammo = 12;
      waterCharges = Math.min(5, waterCharges + 2);
      toast('⚡ طاقة إضافية وماء كامل!');
      syncState();
    } else {
      toast('🪙 تحتاجين إلى 20 عملة لشراء الوجبة');
    }
  }
}

function safeCameraPosition(v) {
  const out = v.clone();
  if (out.y < 1.0) out.y = 1.0;
  for (const obs of cameraObstacles) {
    const pad = .55;
    if (out.x > obs.x - obs.w / 2 - pad && out.x < obs.x + obs.w / 2 + pad &&
        out.z > obs.z - obs.d / 2 - pad && out.z < obs.z + obs.d / 2 + pad &&
        out.y < obs.h + .5) {
      out.y = Math.max(out.y, obs.h + .6);
    }
  }
  return out;
}

function renderMapCanvas(cv, big) {

  ctx.fillStyle = '#3b4359';
  ctx.fillRect(w / 2 - 6.5 * scale, 0, 13 * scale, h);

  const maxR = rect.width * .36;
  let dx = clientX - cx, dy = clientY - cy;

function applyRealisticEnvironment() {
  try {
    const pmrem = new THREE.PMREMGenerator(renderer);
    const envScene = new THREE.Scene();
    envScene.background = new THREE.Color(0x9ec5e8);
    const rt = pmrem.fromScene(envScene, 0.04);
    scene.environment = rt.texture;
    scene.environmentIntensity = 0.45;
    pmrem.dispose();
  } catch (e) {
    console.warn('Environment fallback:', e);
  }
}

function base64ToArrayBufferSafe(base64) {
  if (typeof base64 !== 'string' || base64.length < 32) {
    throw new Error('Invalid or empty base64 string');
  }
  let clean = base64.replace(/[^A-Za-z0-9+/=]/g, '');
  const mod = clean.length % 4;
  if (mod === 2) clean += '==';
  else if (mod === 3) clean += '=';
  const raw = atob(clean);
  const bytes = new Uint8Array(raw.length);
  for (let i = 0; i < raw.length; i++) bytes[i] = raw.charCodeAt(i);
  return bytes.buffer;
}

async function unpackGLB(base64) {
  const buf = base64ToArrayBufferSafe(base64);
  return await new Promise((resolve, reject) => {
    glbLoader.parse(buf, '', gltf => resolve(gltf.scene), reject);
  });
}

function v3SwapMeshes(meshes) {
  if (!Array.isArray(meshes)) return;
  for (const m of meshes) if (m) m.visible = false;
}
function v3AddModel(sample, parent) {
  if (!sample || !parent) return null;
  const model = sample.clone(true);
  parent.add(model);
  return model;
}
function v3RegisterNightWindows(node) {
  if (!node) return;
  node.traverse(o => {
    if (!o.isMesh || !o.material) return;
    const arr = Array.isArray(o.material) ? o.material : [o.material];
    for (const mm of arr) {
      if (!mm) continue;
      if (mm.name === 'windowWarm' || /window|emissive|light/i.test(mm.name || '')) {
        mm.emissive = mm.emissive || new THREE.Color(0x000000);
        mm.emissive.setHex(0xffd7a2);
        v3EmissiveWindows.add(mm);
      }
    }
  });
}
function v3CloneMaterialsSafely(root) {
  if (!root) return;
  root.traverse(o => {
    if (!o.isMesh || !o.material) return;
    o.material = Array.isArray(o.material) ? o.material.map(m => m.clone()) : o.material.clone();
  });
}

async function v3UpgradeModels() {
  if (v3Upgrading || v3UpgradedOnce) return 0;
  const keys = Object.keys(v3EmbeddedGLBs);
  if (!keys.length) return 0;

  v3Upgrading = true;
  applyRealisticEnvironment();
  const loaded = {};
  for (const key of keys) {
    await new Promise(r => setTimeout(r, 45));
    try {
      loaded[key] = await unpackGLB(v3EmbeddedGLBs[key]);
    } catch (e) {
      console.warn('v3 model fallback', key, e);
    }
  }

  let changed = 0;
  for (let i = 0; i < legacyBuildings.length; i++) {
    const b = legacyBuildings[i];
    const type = b.curtain ? 'glass_tower' : (i % 4 === 0 ? 'retail_mall' : i % 3 === 0 ? 'residential_tower' : 'glass_tower');
    const sample = loaded[type]; if (!sample) continue;
    const o = v3AddModel(sample, scene); if (!o) continue;
    o.position.set(b.x, .12, b.z);
    o.scale.set(b.w / 10, b.h / 20, b.d / 10);
    v3SwapMeshes(b.meshes);
    v3RegisterNightWindows(o);
    if (b.name) sign(b.name, b.x, 4.65, b.z - b.d / 2 - 1.0, Math.min(5.4, b.w * .36));
    changed++;
  }

  if (loaded.broadleaf_tree) {
    for (const tr of legacyTrees) {
      const t = v3AddModel(loaded.broadleaf_tree, scene); if (!t) continue;
      t.position.set(tr.x, 0, tr.z);
      t.scale.setScalar((tr.kind === 'potted' ? .78 : .90) * tr.s);
      v3SwapMeshes(tr.meshes);
      changed++;
    }
    if (v3LegacyFoliage) v3LegacyFoliage.visible = false;
  }

  const vehicles = [loaded.urban_sedan, loaded.urban_suv, loaded.urban_sport].filter(Boolean);
  if (vehicles.length) {
    [...legacyParkedCars, ...movingTraffic.map(t => t.group)].forEach((g, i) => {
      const original = g.children.slice();
      const o = v3AddModel(vehicles[i % vehicles.length], g); if (!o) return;
      o.scale.setScalar(.84);
      v3SwapMeshes(original);
      changed++;
    });
  }

  if (loaded.explorer_head && bodyRoot) {
    v3HeadRoot = v3AddModel(loaded.explorer_head, bodyRoot);
    if (v3HeadRoot) {
      v3HeadRoot.position.set(0, 2.27, -.025);
      v3HeadRoot.scale.setScalar(.84);
      v3CloneMaterialsSafely(v3HeadRoot);
      const curve = new THREE.CatmullRomCurve3([
        new THREE.Vector3(-.14, -.20, -.345), new THREE.Vector3(-.085, -.263, -.377),
        new THREE.Vector3(0, -.277, -.389), new THREE.Vector3(.085, -.263, -.377), new THREE.Vector3(.14, -.20, -.345)
      ]);
      const mouth = new THREE.Mesh(new THREE.TubeGeometry(curve, 22, .013, 6, false), new THREE.MeshStandardMaterial({ color: 0x9a5264, roughness: .95 }));
      v3HeadRoot.add(mouth);
      const cheekHighlight = new THREE.MeshBasicMaterial({ color: 0xffc3ce, transparent: true, opacity: .5 });
      for (const s of [-1, 1]) {
        const blushDot = new THREE.Mesh(new THREE.SphereGeometry(.065, 10, 8), cheekHighlight);
        blushDot.position.set(s * .235, -.18, -.328); blushDot.scale.set(1, .4, .21);
        v3HeadRoot.add(blushDot);
      }
      v3SwapMeshes(v3LegacyHead);
      applyLook();
      changed++;
    }
  }

  if (changed > 0) {
    const note = $('v3QualityIndicator');
    if (note) note.textContent = `✓ 3D models: ${changed}`;
    if (active) toast('🏙️ تم تحميل نماذج المدينة الجديدة');
    v3UpgradedOnce = true;
  }
  v3Upgrading = false;
  return changed;
}

let last = performance.now() / 1000, elapsed = 0, mapTimer = 0;
let lastWorldTick = 0, lastLightingTick = 0, lastRenderTick = 0, engineErrors = 0;

const _camTargetPos = new THREE.Vector3();
const _lookTargetPos = new THREE.Vector3();
const _seatWorldVec = new THREE.Vector3();

function animate() {
  requestAnimationFrame(animate);
  const rawNow = performance.now();
  if (rawNow - lastRenderTick < 32) return;
  lastRenderTick = rawNow;

  try {
    const now = rawNow / 1000;
    let dt = Math.min(.045, Math.max(0.001, now - last));
    last = now;
    elapsed += dt;

    if (now - lastLightingTick > .20) {
      updateDayNight();
      lastLightingTick = now;
    }

    if (now - lastWorldTick > .065) {
      const stepDt = Math.min(.12, now - lastWorldTick || dt);
      updateTraffic(stepDt);
      lastWorldTick = now;
    }

    const playing = active && !paused;
    if (playing && !activity) {
      let forward = (keys.has('KeyW') || keys.has('ArrowUp') ? 1 : 0) - (keys.has('KeyS') || keys.has('ArrowDown') ? 1 : 0);
      let right = (keys.has('KeyD') || keys.has('ArrowRight') ? 1 : 0) - (keys.has('KeyA') || keys.has('ArrowLeft') ? 1 : 0);
      forward += joy.y; right += joy.x;
      if (now < voiceMoveUntil) { forward += voiceMove.f; right += voiceMove.r; }
      let l = Math.hypot(forward, right);
      if (l > 1) { forward /= l; right /= l; }

      const isSprint = !!(sprintButton || keys.has('ShiftLeft') || keys.has('ShiftRight') || now < forcedSprintUntil);
      let speed = isSprint ? 10.8 : 7.15;
      if (now < snackBoostUntil) speed *= 1.25;

      let dx = (-Math.sin(cameraYaw) * forward + Math.cos(cameraYaw) * right) * speed * dt;
      let dz = (-Math.cos(cameraYaw) * forward - Math.sin(cameraYaw) * right) * speed * dt;
      if (walkable(pos.x + dx, pos.z)) pos.x += dx;
      if (walkable(pos.x, pos.z + dz)) pos.z += dz;

      let moving = l > .09;
      if (moving && grounded) stepSound(now, isSprint);
      if (moving && (Math.abs(dx) > 1e-4 || Math.abs(dz) > 1e-4)) {
        let desired = Math.atan2(-dx, -dz);
        let delta = Math.atan2(Math.sin(desired - avatar.rotation.y), Math.cos(desired - avatar.rotation.y));
        avatar.rotation.y += delta * Math.min(1, dt * 11);
      }

      verticalVelocity -= 19 * dt;
      pos.y += verticalVelocity * dt;
      if (pos.y <= .12) { pos.y = .12; verticalVelocity = 0; grounded = true; }

      let amp = moving ? Math.min(1, l) * (isSprint ? 0.75 : 0.47) : 0;
      walkPhase += moving ? dt * (isSprint ? 13 : 9) : dt * 2;
      leftLeg.rotation.x = THREE.MathUtils.damp(leftLeg.rotation.x, Math.sin(walkPhase) * amp, 12, dt);
      rightLeg.rotation.x = THREE.MathUtils.damp(rightLeg.rotation.x, -Math.sin(walkPhase) * amp, 12, dt);
      leftArm.rotation.x = THREE.MathUtils.damp(leftArm.rotation.x, -Math.sin(walkPhase) * amp * .85, 12, dt);
      rightArm.rotation.x = THREE.MathUtils.damp(rightArm.rotation.x, Math.sin(walkPhase) * amp * .85, 12, dt);
      bodyRoot.rotation.z = Math.sin(walkPhase) * amp * .048;
      bodyRoot.position.y = grounded ? (moving ? Math.abs(Math.sin(walkPhase)) * .055 : Math.sin(elapsed * 2.1) * .025) : .03;

      for (let g of collectibles) if (g.g.visible && near(pos, g, 1.45)) pickupGem(g);

      let nearItem = nearestInteraction();
      let hint = $('hint');
      if (hint) {
        if (nearItem) {
          hint.textContent = '✋ ' + nearItem.label + ' • زر تفاعل / E';
          hint.classList.add('show');
        } else {
          hint.classList.remove('show');
        }
      }
    }

    swingRig.rotation.x = Math.sin(elapsed * (activity?.kind === 'swing' ? 2.7 : 1.1)) * (activity?.kind === 'swing' ? .43 : .065);
    if (activity) {
      const seated = activity.kind === 'swing';
      if (seated) {
        _seatWorldVec.set(0, -2.18, 0);
        swingRig.localToWorld(_seatWorldVec);
        pos.set(_seatWorldVec.x, _seatWorldVec.y - .98, _seatWorldVec.z);
        avatar.rotation.y = Math.PI;
      } else {
        pos.set(activity.x, -.24, activity.z);
        avatar.rotation.y = 0;
      }
      leftLeg.rotation.x = THREE.MathUtils.damp(leftLeg.rotation.x, .95, 8, dt);
      rightLeg.rotation.x = THREE.MathUtils.damp(rightLeg.rotation.x, .95, 8, dt);
      leftArm.rotation.x = THREE.MathUtils.damp(leftArm.rotation.x, -.36, 8, dt);
      rightArm.rotation.x = THREE.MathUtils.damp(rightArm.rotation.x, -.36, 8, dt);
      bodyRoot.position.y = 0;
      bodyRoot.rotation.z = 0;
      const hint = $('hint');
      if (hint) {
        hint.textContent = '✋ تفاعل أو عصا المشي للنهوض · ↺ إنقاذ';
        hint.classList.add('show');
      }
      footShadow.visible = false;
    } else {
      footShadow.visible = true;
    }

    sittingLegs.visible = !!activity && activity.kind === 'seat';
    leftLeg.visible = rightLeg.visible = !sittingLegs.visible;

    avatar.position.copy(pos);
    footShadow.position.set(pos.x, .155, pos.z);
    footShadow.scale.setScalar(grounded ? 1 : Math.max(.65, 1 - (pos.y - .12) * .13));

    let cx = pos.x + Math.sin(cameraYaw) * cameraDist * Math.cos(cameraPitch);
    let cy = pos.y + 2.1 + Math.sin(cameraPitch) * cameraDist * .68;
    let cz = pos.z + Math.cos(cameraYaw) * cameraDist * Math.cos(cameraPitch);
    _camTargetPos.set(cx, cy, cz);
    _lookTargetPos.set(pos.x, pos.y + 1.35, pos.z);

    cameraSmooth.lerp(safeCameraPosition(_camTargetPos), 1 - Math.exp(-dt * 10));
    lookSmooth.lerp(_lookTargetPos, 1 - Math.exp(-dt * 10));
    cameraSmooth.copy(safeCameraPosition(cameraSmooth));
    camera.position.copy(cameraSmooth);
    camera.lookAt(lookSmooth);
    sky.position.copy(camera.position);
    stars.position.copy(camera.position);

    for (let g of collectibles) {
      if (!g.g.visible) continue;
      g.g.position.y = .95 + Math.sin(elapsed * 2.1 + g.phase) * .19;
      g.crystal.rotation.y += dt * 1.4;
      g.ring.rotation.z += dt * .35;
    }
    if (keyGroup.visible) {
      keyGroup.rotation.y += dt * .9;
      keyGroup.position.y = 2.0 + Math.sin(elapsed * 2.5) * .22;
    }
    for (const fb of flowerBeds) {
      if (fb.watered && fb.blossom.visible) {
        fb.blossom.scale.lerp(_lookTargetPos.set(1.35, 1.35, 1.35), 1 - Math.exp(-dt * 5));
        _lookTargetPos.set(pos.x, pos.y + 1.35, pos.z);
      }
    }
    for (let ch of chests) {
      if (ch.opened) {
        ch.opening = Math.min(1, ch.opening + dt * 1.7);
        ch.lid.rotation.x = 1.6 * (1 - Math.pow(1 - ch.opening, 3));
      } else {
        ch.icon.position.y = 1.8 + Math.sin(elapsed * 2.2 + ch.x) * .15;
        ch.icon.rotation.y += dt;
      }
    }
    if (data.gate && gate.position.y < 9.1) gate.position.y = Math.min(9.1, gate.position.y + dt * 5.6);

    for (const q of pedestrians) {
      let phase = elapsed * q.speed * .28 + q.offset;
      if (q.path === 0) {
        q.p.position.z = q.homeZ + Math.sin(phase) * 5.0;
        q.p.rotation.y = Math.cos(phase) >= 0 ? Math.PI : 0;
      } else {
        q.p.position.x = q.homeX + Math.sin(phase) * 4.5;
        q.p.rotation.y = Math.cos(phase) >= 0 ? -Math.PI / 2 : Math.PI / 2;
      }
      const distP = Math.hypot(q.p.position.x - pos.x, q.p.position.z - pos.z);
      if (distP > 0.05 && distP < 5) {
        const dx = pos.x - q.p.position.x, dz = pos.z - q.p.position.z;
        q.p.rotation.y = Math.atan2(-dx, -dz);
      }
      let sw = Math.sin(elapsed * q.speed * 9 + q.offset) * .37;
      q.limbs[0].rotation.x = sw; q.limbs[1].rotation.x = -sw * .75;
      q.limbs[2].rotation.x = -sw; q.limbs[3].rotation.x = sw * .75;
    }

    fountainWater.material.opacity = .86 + Math.sin(elapsed * 2) * .035;
    for (let d of decorations) {
      if (d.type === 'water') d.mesh.position.y = .85 + Math.sin(elapsed * 4 + d.phase) * .14;
      else if (d.type === 'jet') d.mesh.scale.y = .78 + Math.sin(elapsed * 3 + d.phase) * .17;
    }

    for (let i = particlePieces.length - 1; i >= 0; i--) {
      let p = particlePieces[i];
      p.life -= dt;
      p.mesh.position.addScaledVector(p.velocity, dt);
      p.velocity.y -= 8 * dt;
      p.mesh.rotation.x += dt * 4;
      p.mesh.scale.setScalar(Math.max(.01, p.life / 1.1));
      if (p.life <= 0) {
        scene.remove(p.mesh);
        p.mesh.geometry.dispose();
        p.mesh.material.dispose();
        particlePieces.splice(i, 1);
      }
    }

    mapTimer += dt;
    if (mapTimer > .16) {
      mapTimer = 0;
      if (active) {
        drawMap($('miniMap'), false);
        if (mapOpen) drawMap($('bigMap'), true);
      }
    }

    renderer.render(scene, camera);
    engineErrors = 0;
  } catch (error) {
    engineErrors++;
    console.error('City game frame error:', error);
    if (engineErrors === 1) {
      const el = document.getElementById('toast');
      if (el) {
        el.textContent = '⚠️ حدث خطأ في تشغيل اللعبة؛ جرّبي إعادة فتح الصفحة';
        el.classList.add('show');
      }
    }
    if (engineErrors > 5) {
      active = false;
      paused = true;
    }
  }
}
